// ---------------- Base64 Encoder ----------------
String imgToBase64(camera_fb_t* fb) {
  if (!fb || !fb->buf || fb->len == 0) {
    Serial.println("Invalid frame buffer for base64 encoding!");
    return "";
  }

  size_t output_len = 0;
  if (mbedtls_base64_encode(NULL, 0, &output_len, fb->buf, fb->len) != MBEDTLS_ERR_BASE64_BUFFER_TOO_SMALL) {
    Serial.println("Failed to get base64 output length");
    return "";
  }

  unsigned char* output = (unsigned char*)malloc(output_len + 1);
  if (!output) {
    Serial.println("Memory allocation failed");
    return "";
  }

  if (mbedtls_base64_encode(output, output_len, &output_len, fb->buf, fb->len) != 0) {
    Serial.println("Base64 encoding failed");
    free(output);
    return "";
  }
  output[output_len] = '\0';

  String base64Str((char*)output);
  free(output);

  Serial.printf("Base64 length: %d\n", (int)output_len);
  Serial.print("Base64 sample: ");
  for (int i = 0; i < min(50, (int)output_len); i++) Serial.print(base64Str[i]);
  Serial.println();

  return base64Str;
}

// ---------------- Upload to ImgBB ----------------
String uploadToImgBB(camera_fb_t * fb) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi not connected for image upload");
    return "";
  }

  String base64Img = imgToBase64(fb);
  if (base64Img.length() == 0) {
    Serial.println("Base64 encoding failed!");
    return "";
  }

  base64Img.replace("\n", "");
  base64Img.replace("\r", "");

  HTTPClient http;
  String url = "https://api.imgbb.com/1/upload?key=" + String(IMGBB_API_KEY);
  http.begin(url);

  String boundary = "----ESP32CAMBoundary";
  String bodyStart = "--" + boundary + "\r\n";
  bodyStart += "Content-Disposition: form-data; name=\"image\"\r\n\r\n";
  String bodyEnd = "\r\n--" + boundary + "--\r\n";

  String fullBody = bodyStart + base64Img + bodyEnd;

  http.addHeader("Content-Type", "multipart/form-data; boundary=" + boundary);
  http.addHeader("Accept", "application/json");

  int httpResponseCode = http.POST(fullBody);
  String payload = http.getString();

  if (httpResponseCode == 200) {
    Serial.println("ImgBB Response: " + payload);
    int urlIndex = payload.indexOf("\"url\":\"");
    if (urlIndex > 0) {
      int start = urlIndex + 7;
      int end = payload.indexOf("\"", start);

      String cleanURL = payload.substring(start, end);
      cleanURL.replace("\\/", "/");

      http.end();
      // return payload.substring(start, end);
      return cleanURL;

    }
  } else {
    Serial.printf("ImgBB POST failed: %d\nPayload: %s\n", httpResponseCode, payload.c_str());
  }

  http.end();
  return "";
}

// ---------------- Save to Firebase ----------------
bool sendURLToFirebase(const String& url) {
  String imagePath = "/esp32cam/latest_image";
  if (Firebase.RTDB.setString(&fbdo, imagePath, url)) {
    Serial.println("Image URL saved to Firebase: " + imagePath);
    return true;
  } else {
    Serial.println("Firebase error: " + fbdo.errorReason());
    return false;
  }
}

// ---------------- Main ----------------
void setup() {
  Serial.begin(115200);
  delay(1000);

  printHeapStats();
  connectWiFi();

  if (!setupCamera()) {
    Serial.println("Camera setup failed. Restarting.");
    delay(2000);
    ESP.restart();
  }

  fbConfig.api_key = API_KEY;
  fbConfig.database_url = DATABASE_URL;

  // ✅ Anonymous login (important fix)
  if (Firebase.signUp(&fbConfig, &fbAuth, "", "")) {
    Serial.println("Firebase anonymous login successful!");
  } else {
    Serial.printf("Firebase signUp failed: %s\n", fbConfig.signer.signupError.message.c_str());
  }

  Firebase.begin(&fbConfig, &fbAuth);
  Firebase.reconnectWiFi(true);

  Serial.println("Setup complete!");
}

void loop() {
  printHeapStats();

  camera_fb_t * fb = esp_camera_fb_get();
  if (!fb) {
    Serial.println("Camera capture failed, retrying...");
    delay(3000);
    return;
  }

  String imgURL = uploadToImgBB(fb);
  esp_camera_fb_return(fb);

  if (imgURL.length() > 0) {
    Serial.println("Uploaded Image URL: " + imgURL);
    if (!sendURLToFirebase(imgURL)) {
      Serial.println("Failed to upload image URL to Firebase, retrying next cycle.");
    }
  } else {
    Serial.println("Image upload failed, will retry.");
  }

  delay(15000); // 15s between captures
}
