#include <WiFi.h>
#include <HTTPClient.h>
#include <Firebase_ESP_Client.h>
#include "esp_camera.h"

extern "C" {
  #include "mbedtls/base64.h"
}

// ---------------- USER SETTINGS ----------------
#define WIFI_SSID       ""   // <-- add WiFi SSID
#define WIFI_PASSWORD   ""   // <-- add WiFi password

#define IMGBB_API_KEY   ""   // <-- add ImgBB API key
#define API_KEY         ""   // <-- add Firebase Web API key
#define DATABASE_URL    ""   // <-- add Firebase Realtime DB URL

// Choose resolution: 0 = QVGA (safe), 1 = VGA (higher but riskier)
#define USE_VGA 0

// Capture interval (ms)
#define CAPTURE_INTERVAL 15000

// ---------------- Firebase ----------------
FirebaseData fbdo;
FirebaseAuth fbAuth;
FirebaseConfig fbConfig;

// ---------------- Camera Pins (AI Thinker ESP32-CAM) ----------------
#define PWDN_GPIO_NUM     -1
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM      0
#define SIOD_GPIO_NUM     26
#define SIOC_GPIO_NUM     27
#define Y9_GPIO_NUM       35
#define Y8_GPIO_NUM       34
#define Y7_GPIO_NUM       39
#define Y6_GPIO_NUM       36
#define Y5_GPIO_NUM       21
#define Y4_GPIO_NUM       19
#define Y3_GPIO_NUM       18
#define Y2_GPIO_NUM        5
#define VSYNC_GPIO_NUM    25
#define HREF_GPIO_NUM     23
#define PCLK_GPIO_NUM     22

// ---------------- Helpers ----------------
void printHeapStats(const char* tag = "") {
  Serial.printf("[HEAP] %s Free Heap: %d | Free PSRAM: %d\n",
                tag, ESP.getFreeHeap(), ESP.getFreePsram());
}

void connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("[WiFi] Connecting");
  for (int i = 0; i < 60 && WiFi.status() != WL_CONNECTED; i++) {
    delay(250);
    Serial.print(".");
  }
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("\n[WiFi] Failed. Restarting...");
    delay(1500);
    ESP.restart();
  }
  Serial.printf("\n[WiFi] Connected. IP: %s\n", WiFi.localIP().toString().c_str());
}

bool setupCamera() {
  camera_config_t config;
  config.ledc_timer   = LEDC_TIMER_0;
  config.ledc_channel = LEDC_CHANNEL_0;

  config.pin_d0       = Y2_GPIO_NUM;
  config.pin_d1       = Y3_GPIO_NUM;
  config.pin_d2       = Y4_GPIO_NUM;
  config.pin_d3       = Y5_GPIO_NUM;
  config.pin_d4       = Y6_GPIO_NUM;
  config.pin_d5       = Y7_GPIO_NUM;
  config.pin_d6       = Y8_GPIO_NUM;
  config.pin_d7       = Y9_GPIO_NUM;
  config.pin_xclk     = XCLK_GPIO_NUM;
  config.pin_pclk     = PCLK_GPIO_NUM;
  config.pin_vsync    = VSYNC_GPIO_NUM;
  config.pin_href     = HREF_GPIO_NUM;
  config.pin_sscb_sda = SIOD_GPIO_NUM;
  config.pin_sscb_scl = SIOC_GPIO_NUM;
  config.pin_pwdn     = PWDN_GPIO_NUM;
  config.pin_reset    = RESET_GPIO_NUM;

  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_JPEG;

#if USE_VGA
  config.frame_size   = FRAMESIZE_VGA;   // 640x480
  config.jpeg_quality = 42;              // higher = smaller
#else
  config.frame_size   = FRAMESIZE_QVGA;  // 320x240
  config.jpeg_quality = 35;              // balanced
#endif
  config.fb_count     = 1;               // one buffer → saves RAM

  if (esp_camera_init(&config) != ESP_OK) {
    Serial.println("[CAM] Init failed!");
    return false;
  }
  Serial.println("[CAM] Initialized.");
  return true;
}

// ---------------- Base64 Encoder ----------------
String imgToBase64(camera_fb_t* fb) {
  if (!fb || !fb->buf || fb->len == 0) {
    Serial.println("[B64] Invalid frame!");
    return "";
  }

  size_t out_len = 0;
  if (mbedtls_base64_encode(NULL, 0, &out_len, fb->buf, fb->len) != MBEDTLS_ERR_BASE64_BUFFER_TOO_SMALL) {
    Serial.println("[B64] Length calc failed");
    return "";
  }

  unsigned char* out = (unsigned char*)malloc(out_len + 1);
  if (!out) {
    Serial.println("[B64] malloc failed");
    return "";
  }

  if (mbedtls_base64_encode(out, out_len, &out_len, fb->buf, fb->len) != 0) {
    Serial.println("[B64] encode failed");
    free(out);
    return "";
  }
  out[out_len] = '\0';

  String base64Str;
  base64Str.reserve(out_len + 1);
  base64Str = (char*)out;
  free(out);

  Serial.printf("[B64] len=%d (jpeg=%d)\n", (int)out_len, (int)fb->len);
  return base64Str;
}

// ---------------- Upload to ImgBB ----------------
String uploadToImgBB(camera_fb_t* fb) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("[HTTP] No WiFi");
    return "";
  }

  String b64 = imgToBase64(fb);
  if (b64.isEmpty()) return "";

  b64.replace("\n", ""); b64.replace("\r", "");

  HTTPClient http;
  String url = "https://api.imgbb.com/1/upload?key=" + String(IMGBB_API_KEY);
  if (!http.begin(url)) return "";

  const String boundary = "----ESP32CAMBoundary";
  String body = "--" + boundary + "\r\n"
                "Content-Disposition: form-data; name=\"image\"\r\n\r\n" +
                b64 + "\r\n--" + boundary + "--\r\n";

  http.addHeader("Content-Type", "multipart/form-data; boundary=" + boundary);
  http.addHeader("Accept", "application/json");

  printHeapStats("pre-POST");
  int code = http.POST(body);
  String payload = http.getString();
  http.end();

  if (code != 200) {
    Serial.printf("[HTTP] POST failed: %d\nPayload: %s\n", code, payload.c_str());
    return "";
  }

  int idx = payload.indexOf("\"url\":\"");
  if (idx < 0) return "";
  int start = idx + 7;
  int end   = payload.indexOf("\"", start);
  if (end <= start) return "";

  String cleanURL = payload.substring(start, end);
  cleanURL.replace("\\/", "/");
  Serial.println("[HTTP] ImgBB URL: " + cleanURL);
  return cleanURL;
}

// ---------------- Save to Firebase ----------------
bool sendURLToFirebase(const String& url) {
  const String path = "/esp32cam/latest_image";
  if (Firebase.RTDB.setString(&fbdo, path, url)) {
    Serial.println("[FB] Saved URL");
    return true;
  }
  Serial.println("[FB] Error: " + fbdo.errorReason());
  return false;
}

// ---------------- Main ----------------
void setup() {
  Serial.begin(115200);
  delay(400);

  printHeapStats("boot");
  connectWiFi();

  if (!setupCamera()) {
    delay(2000);
    ESP.restart();
  }

  fbConfig.api_key      = API_KEY;
  fbConfig.database_url = DATABASE_URL;

  if (Firebase.signUp(&fbConfig, &fbAuth, "", "")) {
    Serial.println("[FB] Anonymous login OK");
  } else {
    Serial.printf("[FB] signUp failed: %s\n", fbConfig.signer.signupError.message.c_str());
  }

  Firebase.begin(&fbConfig, &fbAuth);
  Firebase.reconnectWiFi(true);
  Serial.println("[BOOT] Setup complete");
}

void loop() {
  printHeapStats("loop");

  camera_fb_t* fb = esp_camera_fb_get();
  if (!fb) {
    Serial.println("[CAM] Capture failed");
    delay(3000);
    return;
  }

  String url = uploadToImgBB(fb);
  esp_camera_fb_return(fb); // free buffer
  fb = NULL;

  if (url.length()) sendURLToFirebase(url);

  yield(); // allow cleanup
  delay(CAPTURE_INTERVAL);
}
