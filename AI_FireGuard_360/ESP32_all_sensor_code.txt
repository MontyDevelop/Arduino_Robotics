#include <WiFi.h>
#include <Firebase_ESP_Client.h>
#include <DHT.h>

// ----------------- WiFi Credentials -----------------
#define WIFI_SSID ""
#define WIFI_PASSWORD ""

// ----------------- Firebase Config -----------------
#define DATABASE_URL ""  // Your RTDB URL
#define API_KEY ""  // Copy from Firebase Console → Project Settings → General → Web API Key

// ---------------- Firebase Objects ----------------
FirebaseData fbdo;
FirebaseAuth auth;
FirebaseConfig config;

// ---------------- Sensors ----------------
#define DHTPIN 4        // DHT11 pin
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

#define MQ2_PIN 34      // MQ2 analog pin
#define MQ135_PIN 35    // MQ135 analog pin
#define FLAME_PIN 26    // Flame digital pin
#define PIR_PIN 27      // PIR digital pin

// ---------------- Setup ----------------
void setup() {
  Serial.begin(115200);
  pinMode(PIR_PIN, INPUT);
  dht.begin();

  // WiFi Connect
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nConnected! IP: " + WiFi.localIP().toString());

  // Firebase Setup
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  // Anonymous login
  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("Firebase signUp ok");
  } else {
    Serial.printf("Firebase signUp failed: %s\n", config.signer.signupError.message.c_str());
  }

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
}

// ---------------- Loop ----------------
void loop() {
  // Read DHT11
  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  // Read gas sensors
  int mq2Value = analogRead(MQ2_PIN);
  int mq135Value = analogRead(MQ135_PIN);
  int flame = digitalRead(FLAME_PIN); 
  // Note: LOW = flame detected, HIGH = no flame

  // Read PIR
  int motion = digitalRead(PIR_PIN);

  // Rule-based AI (simple thresholds)
  String risk = "LOW";
  if (temperature > 50 || mq2Value > 2000 || mq135Value > 2000 || motion == HIGH) {
    risk = "HIGH";
  } else if (temperature > 35 || mq2Value > 1000 || mq135Value > 1000) {
    risk = "MEDIUM";
  }

  // Print data
  Serial.println("---- Sensor Data ----");
  Serial.printf("Temp: %.1f °C  Hum: %.1f %%\n", temperature, humidity);
  Serial.printf("MQ2: %d  MQ135: %d  PIR: %d\n", mq2Value, mq135Value, motion);
  Serial.printf("Flame: %d\n", flame);
  Serial.println("Risk Status: " + risk);

  // Upload to Firebase
  if (Firebase.ready()) {
    Firebase.RTDB.setFloat(&fbdo, "/sensors/temperature", temperature);
    Firebase.RTDB.setFloat(&fbdo, "/sensors/humidity", humidity);
    Firebase.RTDB.setInt(&fbdo, "/sensors/mq2", mq2Value);
    Firebase.RTDB.setInt(&fbdo, "/sensors/mq135", mq135Value);
    Firebase.RTDB.setInt(&fbdo, "/sensors/flame", flame);
    Firebase.RTDB.setInt(&fbdo, "/sensors/motion", motion);
    Firebase.RTDB.setString(&fbdo, "/sensors/risk_status", risk);
  }

  delay(2000); // wait 5 sec
}
