#include <SoftwareSerial.h>

#define SOIL_MOISTURE_PIN A0  // Soil moisture sensor connected to A0

SoftwareSerial esp8266(2, 3); // RX, TX (connect to ESP8266)

String apiKey = "your-api-key";  // API key for the IoT platform
const char* ssid = "your-SSID";  // Replace with your WiFi SSID
const char* password = "your-password";  // Replace with your WiFi password
const char* server = "api.example.com";  // Replace with your IoT server URL
int moistureValue = 0;

void setup() {
  Serial.begin(9600);
  esp8266.begin(9600);  // Start communication with ESP8266
  pinMode(SOIL_MOISTURE_PIN, INPUT);

  connectWiFi();
}

void loop() {
  moistureValue = analogRead(SOIL_MOISTURE_PIN);  // Read the sensor
  sendDataToServer(moistureValue);  // Send the data
  delay(60000);  // Send data every 60 seconds
}

void connectWiFi() {
  Serial.println("Connecting to WiFi...");
  esp8266.println("AT");
  delay(1000);
  esp8266.println("AT+CWJAP=\"" + String(ssid) + "\",\"" + String(password) + "\"");
  delay(5000);
  Serial.println("Connected to WiFi");
}

void sendDataToServer(int moistureValue) {
  String data = "moisture=" + String(moistureValue) + "&api_key=" + apiKey;

  esp8266.println("AT+CIPSTART=\"TCP\",\"" + String(server) + "\",80");
  delay(2000);
  
  String request = "GET /update?" + data + " HTTP/1.1\r\n";
  request += "Host: " + String(server) + "\r\n";
  request += "Connection: close\r\n\r\n";

  esp8266.print("AT+CIPSEND=");
  esp8266.println(request.length());
  delay(1000);
  
  esp8266.print(request);
  delay(2000);
  
  Serial.println("Data Sent: " + data);
}
